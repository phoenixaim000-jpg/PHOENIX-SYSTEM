//| 取引レート更新（ティック）毎に実行されるメイン関数                 |
//+------------------------------------------------------------------+
void OnTick()
  {
   // 現在の通貨ペアと価格を取得
   string symbol = _Symbol;
   double ask = SymbolInfoDouble(symbol, SYMBOL_ASK);
   double bid = SymbolInfoDouble(symbol, SYMBOL_BID);
   
   // --- チャンスゾーン（どこからどこまでがレンジか）をAIが自動計算
   int highest_index = iHighest(symbol, TIMEFRAME_M1, MODE_HIGH, InpRangePeriod, 1);
   int lowest_index  = iLowest(symbol, TIMEFRAME_M1, MODE_LOW, InpRangePeriod, 1);
   
   if(highest_index < 0 || lowest_index < 0) return;
   
   double range_high = iHigh(symbol, TIMEFRAME_M1, highest_index); // ゾーン上限（どこまで）
   double range_low  = iLow(symbol, TIMEFRAME_M1, lowest_index);   // ゾーン下限（どこから）

   // 既に持っているポジションの数を確認
   int total_positions = PositionsTotal();
   bool has_position = false;
   
   for(int i=total_positions-1; i>=0; i--)
     {
      if(PositionGetSymbol(i) == symbol)
        {
         if(PositionGetInteger(POSITION_MAGIC) == InpMagicNumber)
           {
            has_position = true;
            
            // --- 【AI決済ロジック】 ---
            // ロングを持っていて上限（どこまで）に達したら、誰かが決済に走る前に利益確定
            if(PositionGetInteger(POSITION_TYPE) == POSITION_TYPE_BUY && bid >= range_high)
              {
               Print("🎯 VISION AI：レンジ上限に達したためロングを決済します。");
               ClosePosition(PositionGetInteger(POSITION_TICKET));
              }
            // ショートを持っていて下限（どこから）に達したら、利益確定決済
            if(PositionGetInteger(POSITION_TYPE) == POSITION_TYPE_SELL && ask <= range_low)
              {
               Print("🎯 VISION AI：レンジ下限に達したためショートを決済します。");
               ClosePosition(PositionGetInteger(POSITION_TICKET));
              }
           }
        }
     }

   // --- 【AIエントリーロジック】 ---
   // ノーポジの時、他がロングや決済に走った歪みをチャンスと捉えて逆張り対応
   if(!has_position)
     {
      // 価格が下限（どこから）を破る、またはタッチした瞬間に「反発ロング」
      if(ask <= range_low)
        {
         Print("💥 VISION AI：レンジ下限に到達。ロング・エントリーを実行。");
         SendMarketOrder(TRADE_ACTION_DEAL, ORDER_TYPE_BUY, ask);
        }
      // 価格が上限（どこまで）に達した瞬間に「反落ショート」
      else if(bid >= range_high)
        {
         Print("💥 VISION AI：レンジ上限に到達。ショート・エントリーを実行。");
         SendMarketOrder(TRADE_ACTION_DEAL, ORDER_TYPE_SELL, bid);
        }
     }
  }

//+------------------------------------------------------------------+
//| 注文送信サブ関数                                                 |
//+------------------------------------------------------------------+
void SendMarketOrder(ENUM_TRADE_REQUEST_ACTIONS action, ENUM_ORDER_TYPE type, double price)
  {
   MqlTradeRequest request={0};
   MqlTradeResult  result={0};
   
   request.action       = action;
   request.symbol       = _Symbol;
   request.volume       = InpLotSize;
   request.type         = type;
   request.price        = price;
   request.deviation    = 20;
   request.magic        = InpMagicNumber;
   request.comment      = "VISION FX AI";
   request.type_filling = ORDER_FILLING_FOK;
   
   if(!OrderSend(request, result))
     {
      Print("注文エラー: ", GetLastError());
     }
  }

//+------------------------------------------------------------------+
//| ポジション決済サブ関数                                            |
//+------------------------------------------------------------------+
void ClosePosition(ulong ticket)
  {
   MqlTradeRequest request={0};
   MqlTradeResult  result={0};
   
   if(PositionSelectByTicket(ticket))
     {
      request.action       = TRADE_ACTION_DEAL;
      request.position     = ticket;
      request.symbol       = PositionGetString(POSITION_SYMBOL);
      request.volume       = PositionGetDouble(POSITION_VOLUME);
      request.deviation    = 20;
      request.magic        = InpMagicNumber;
      
      if(PositionGetInteger(POSITION_TYPE) == POSITION_TYPE_BUY)
        {
         request.type = ORDER_TYPE_SELL;
         request.price = SymbolInfoDouble(request.symbol, SYMBOL_BID);
        }
      else
        {
         request.type = ORDER_TYPE_BUY;
         request.price = SymbolInfoDouble(request.symbol, SYMBOL_ASK);
        }
        
      if(!OrderSend(request, result))
        {
         Print("決済エラー: ", GetLastError());
        }
     }
  }

class VisionFxAiEa:
    def __init__(self):
        self.base_stop_loss = 1.0   # 通常時の損切り幅
        self.tight_stop_loss = 0.5  # 急変時の損切り幅
        
        # 急落・急騰を判定する閾値（1分間で価格が0.2%以上動いたら異常事態とみなす）
        self.volatility_threshold = 0.002 

    def check_market_shock(self, current_price, price_1min_ago):
        """
        1分足の価格差からバブル・急落を検知し、適切な損切り幅を返す関数
        """
        # 1分間での価格の変動率を計算
        price_change_rate = abs(current_price - price_1min_ago) / price_1min_ago
        
        # 閾値を超えた突発的な動き（急騰・急落）がある場合
        if price_change_rate >= self.volatility_threshold:
            print(f"【⚠️ALERT】1分足での急激な変動を検知（変動率: {price_change_rate:.4f}）")
            print(f"バブル・急落対応モード発動：損切り幅を {self.base_stop_loss} から {self.tight_stop_loss} へ引き下げます。")
            return self.tight_stop_loss
        
        # 通常相場の場合
        return self.base_stop_loss

    def execute_trade_order(self, current_price, price_1min_ago, order_type="BUY"):
        """
        動的損切りを適用して注文をシミュレーションする関数
        """
        # 動的な損切り幅を取得
        active_sl_width = self.check_market_shock(current_price, price_1min_ago)
        
        # 実際の損切り価格の計算
        if order_type == "BUY":
            sl_price = current_price - active_sl_width
        else:
            sl_price = current_price + active_sl_width
            
        print(f"【注文実行】確定損切り価格: {sl_price} (幅: {active_sl_width})\n")
        return sl_price
