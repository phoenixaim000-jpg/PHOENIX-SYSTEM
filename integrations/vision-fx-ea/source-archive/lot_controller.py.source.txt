import math

class DynamicLotController:
    def __init__(self, base_lot=1.0, max_lot_multiplier=2.0, min_lot_multiplier=0.1):
        """
        システム初期化
        :param base_lot: 基準となるロット数（通常時のロット）
        :param max_lot_multiplier: 確信度が最高の場合の最大ロット倍率（上限キャップ）
        :param min_lot_multiplier: 自信がない場合の最小ロット倍率（下限キャップ）
        """
        self.base_lot = base_lot
        self.max_lot_multiplier = max_lot_multiplier
        self.min_lot_multiplier = min_lot_multiplier

    def calculate_confidence_score(self, market_conditions):
        """
        相場環境からAIの確信度を計算するロジック
        ※ ここにMovaロジックやインジケーターの判定を統合していきます。
        """
        score = 0.5  # スタートは50% (ニュートラル)

        # 条件1: トレンドの一致度
        if market_conditions.get("trend_aligned") == True:
            score += 0.25
        else:
            score -= 0.15

        # 条件2: ボラティリティ（荒れ相場時は自信を下げる）
        if market_conditions.get("high_volatility") == True:
            score -= 0.15
        else:
            score += 0.10

        # 条件3: 過去の優位性パターン（確信パターン）に入っているか
        if market_conditions.get("signature_pattern_matched") == True:
            score += 0.30

        # スコアを 0.0 ~ 1.0 の範囲に収める
        score = max(0.0, min(1.0, score))
        return score

    def determine_lot_size(self, confidence_score):
        """
        確信度スコアに基づいて最終的な発注ロット数を決定する（境界監査バリア）
        """
        print(f"[AI分析] 現在の確信度スコア: {confidence_score:.2%}")

        # 1. 確信度に応じた倍率の決定
        if confidence_score >= 0.85:
            multiplier = self.max_lot_multiplier
            status = "🔥【確信・最高パターン】ロットを引き上げます"
        elif confidence_score >= 0.60:
            multiplier = 1.0
            status = "👍【通常・期待値あり】基準ロットでエントリーします"
        elif confidence_score >= 0.40:
            multiplier = 0.5
            status = "⚠️【警戒・自信なし】ロットを半分に縮小します"
        else:
            multiplier = 0.0
            status = "🛑【見送り・不確実】リスクが高いためエントリーを回避します"

        print(f"[判定結果] {status}")

        # 2. ロット数の計算と絶対上限・下限の適用
        calculated_lot = self.base_lot * multiplier

        if calculated_lot > 0:
            max_absolute_lot = self.base_lot * self.max_lot_multiplier
            min_absolute_lot = self.base_lot * self.min_lot_multiplier
            
            calculated_lot = max(min_absolute_lot, min(max_absolute_lot, calculated_lot))
            calculated_lot = round(calculated_lot, 2)

        return calculated_lot

"""
================================================================================
VISION FX AI EA - ADVANCED LOT & RISK CONTROLLER (SEED BARRIER)
Target File: lot_controller.py
Description: 政府の資金供給パターンに応じてロット数と防衛障壁（SEED Barrier）を
             動的に自動変動させ、いかなる状況でも資金を安泰に守り抜くコアモジュール。
================================================================================
"""

class GovernmentMode:
    BULK_INJECTION = "BULK_INJECTION"       # 政府が大きくお金を出した時
    NO_INJECTION = "NO_INJECTION"           # 政府がお金を出さなかった時
    FRAGMENTED = "FRAGMENTED"               # 政府が分けて（小出しに）出してきた時

class VisionFxRiskController:
    def __init__(self, target_reserve_ratio: float = 0.40):
        """
        リスクコントローラーの初期化
        :param target_reserve_ratio: 利益から「絶対聖域積立金」に隔離する割合 (デフォルト40%)
        """
        self.target_reserve_ratio = target_reserve_ratio
        # 初期状態は最も安全な「不給付・絶対防衛モード」に設定
        self.current_mode = GovernmentMode.NO_INJECTION
        self.seed_barrier_active = True

    def update_scenario_mode(self, mode: str) -> str:
        """
        政府の資金供給動向に合わせて、システムモードを限界突破切り替えする
        """
        if mode in [GovernmentMode.BULK_INJECTION, GovernmentMode.NO_INJECTION, GovernmentMode.FRAGMENTED]:
            self.current_mode = mode
            return f"【SEED Barrier】システムモードを [{mode}] に正常に切り替えました。完遂安泰稼働中。"
        else:
            self.current_mode = GovernmentMode.NO_INJECTION
            return "【警告】未定義の動向を検知。安全のため『絶対防衛モード』を維持します。"

    def calculate_adaptive_lot(self, balance: float, market_volatility: float, base_risk_pct: float = 0.01) -> float:
        """
        現在の政府モードと市場のボラティリティから、絶対に困らない安全なロット（ポジションサイズ）を自動計算する
        """
        # 基本となる1%リスクモデルロット
        base_lot = (balance * base_risk_pct) / 100000
        
        if self.current_mode == GovernmentMode.BULK_INJECTION:
            # 政府の一括大量供給：流動性が高いためレバレッジ上限を拡大し利益を最大化
            leverage_limit = 3.0
            adjusted_lot = base_lot * leverage_limit * (1.0 / (market_volatility + 0.1))
            
        elif self.current_mode == GovernmentMode.NO_INJECTION:
            # 政府の不給付（静観）：市場収縮に備え、ロットを通常の極小値に抑えて絶対安全を担保
            leverage_limit = 0.5
            adjusted_lot = base_lot * leverage_limit * 0.5
            
        elif self.current_mode == GovernmentMode.FRAGMENTED:
            # 政府の断続的小出し：乱高下ノイズに対応するため、段階的トラップ用にポジションを細分化
            leverage_limit = 1.5
            adjusted_lot = base_lot * leverage_limit * 0.7
        else:
            adjusted_lot = base_lot * 0.1
            
        # 最小ロット0.01を下回らないように丸める
        return max(0.01, round(adjusted_lot, 2))

    def get_max_drawdown_limit(self) -> float:
        """
        現在のモードに応じた「強制遮断（損切）ドローダウン許容値」を返す
        """
        if self.current_mode == GovernmentMode.BULK_INJECTION:
            return 0.08  # トレンドの押し目を考慮し最大8%まで許容
        elif self.current_mode == GovernmentMode.NO_INJECTION:
            return 0.02  # わずか2%のドローダウンでシステムを物理遮断する鉄壁の防御
        elif self.current_mode == GovernmentMode.FRAGMENTED:
            return 0.04  # 往復ビンタを防を防ぐため4%に設定
        return 0.02

    def process_profit_allocation(self, total_profit: float):
        """
        獲得した運用益から「絶対聖域積立金」への配当分と、次の運用残高への追加分を自動計算する
        """
        if total_profit <= 0:
            return 0.0, 0.0
            
        allocated_to_pool = total_profit * self.target_reserve_ratio
        actual_reinvestment = total_profit - allocated_to_pool
        return actual_reinvestment, allocated_to_pool
